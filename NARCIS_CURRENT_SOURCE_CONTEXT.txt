            fetch(`${API_BASE}/api/dashboard/stats`),
            fetch(`${API_BASE}/api/dashboard/recent-cases`),
            fetch(`${API_BASE}/api/dashboard/activity`),
            fetch(`${API_BASE}/api/dashboard/analysis`),
          ]);

        if (
          !statsResponse.ok ||
          !casesResponse.ok ||
          !activityResponse.ok
        ) {
          throw new Error("Dashboard API request failed");
        }

        const statsData: DashboardStats =
          await statsResponse.json();

        const casesData: RecentCase[] =
          await casesResponse.json();

        const activityData: ActivityItem[] =
          await activityResponse.json();

        const analysisResponse = await fetch(
          `${API_BASE}/api/dashboard/analysis`
        );

        if (!analysisResponse.ok) {
          throw new Error("Analysis API request failed");
        }

        const analysisData: AnalysisData =
          await analysisResponse.json();

        setStats(statsData);
        setRecentCases(casesData);
        setActivities(activityData);
        setAnalysis(analysisData);
      } catch (error) {
        console.error("NARCIS dashboard error:", error);
        setApiError(
          "Unable to load live investigation data."
        );
      } finally {
        setLoading(false);
      }
    };

    loadDashboard();

"use client";

import { useEffect, useState } from "react";
import styles from "./login.module.css";

export default function LoginPage() {
  const [visible, setVisible] = useState(false);
  const [authenticating, setAuthenticating] = useState(false);
  const [authenticated, setAuthenticated] = useState(false);
  const [transitioning, setTransitioning] = useState(false);
  const [officerId, setOfficerId] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");

  useEffect(() => {
    const timer = window.setTimeout(() => {
      setVisible(true);
    }, 120);

    return () => window.clearTimeout(timer);
  }, []);

  const handleAuthenticate = async () => {
    if (authenticating || authenticated) {
      return;
    }

    setError("");

    const officerInput = document.getElementById(
      "officer-id"
    ) as HTMLInputElement | null;

    const passwordInput = document.getElementById(
      "password"
    ) as HTMLInputElement | null;

    const currentOfficerId = officerInput?.value.trim() || "";
    const currentPassword = passwordInput?.value || "";

    if (!currentOfficerId || !currentPassword) {
      setError("Please enter your Officer ID and password.");
      return;
    }

    setAuthenticating(true);

    try {
      const response = await fetch(
        "http://127.0.0.1:8000/api/auth/login",
        {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
          },
          body: JSON.stringify({
            officer_id: currentOfficerId,
            password: currentPassword,
          }),
        }
      );

      const data = await response.json();

      if (!response.ok) {
        throw new Error(
          data.detail || "Authentication failed."
        );
      }

      setAuthenticated(true);

      window.setTimeout(() => {
        setTransitioning(true);
      }, 2200);

      window.setTimeout(() => {
        window.location.href = "/dashboard";
      }, 3250);
    } catch (error) {
      setError(
        error instanceof Error
          ? error.message
          : "Unable to connect to the NARCIS authentication server."
      );
      setAuthenticating(false);
    }
  };

  return (
    <main className={`${styles.page} ${visible ? styles.visible : ""} ${transitioning ? styles.transitioning : ""}`}>

      <div className={styles.atmosphere} />
      <div className={styles.grid} />
      <div className={styles.gridGlow} />
      <div className={styles.particles} />

      <div className={`${styles.telemetry} ${styles.telemetryLeft}`}>
        <span>NARCIS // SECURE NETWORK</span>
        <span>NODE 07</span>
        <span>ENCRYPTED CHANNEL</span>
          <span>LINK STATUS <b>● ONLINE</b></span>
      </div>

      <div className={`${styles.telemetry} ${styles.telemetryRight}`}>
        <span>SYSTEM STATUS</span>
        <span>CASE DATABASE <b>READY</b></span>
        <span>RISK ENGINE <b>READY</b></span>
        <span>RAG CORE <b>READY</b></span>
      </div>

      <div className={`${styles.corner} ${styles.cornerTL}`} />
      <div className={`${styles.corner} ${styles.cornerTR}`} />
      <div className={`${styles.corner} ${styles.cornerBL}`} />
      <div className={`${styles.corner} ${styles.cornerBR}`} />

      <section className={styles.content}>

        <div className={styles.brand}>

          <div className={styles.logoWrap}>
            <div className={styles.logoGlow} />

            <img
              src="/narcis-logo.png"
              alt="NARCIS"
              className={styles.logo}
            />
          </div>

          <div className={styles.systemName}>
            NARCIS INTELLIGENCE SYSTEM
          </div>

          <div className={styles.systemLine}>
            <span />
            <small>SECURE INVESTIGATION ENVIRONMENT</small>
            <span />
          </div>

        </div>

        <div className={`${styles.panel} ${authenticated ? styles.success : ""}`}>

          <div className={styles.panelTopline} />

          <div className={styles.status}>
            <span className={styles.statusLight} />
            {authenticated ? "ACCESS VERIFIED" : "SECURE ACCESS"}
          </div>

          <h1>
            {authenticated ? "ACCESS VERIFIED" : "OFFICER LOGIN"}
          </h1>

          <p className={styles.intro}>
            {authenticated
              ? "Secure authentication established. Initializing the NARCIS investigation environment."
              : "Authenticate to access the NARCIS investigation intelligence environment."}
          </p>

          {!authenticated && (
            <>
              <div className={styles.field}>
                <label htmlFor="officer-id">
                  OFFICER ID
                </label>

                <div className={styles.inputWrap}>
                  <span className={styles.inputMarker} />

                  <input
                    id="officer-id"
                    type="text"
                    placeholder="Enter officer ID"
                    autoComplete="username"
                  />
                </div>
              </div>

              <div className={styles.field}>
                <label htmlFor="password">
                  PASSWORD
                </label>

                <div className={styles.inputWrap}>
                  <span className={styles.inputMarker} />

                  <input
                    id="password"
                    type="password"
                    placeholder="Enter password"
                    autoComplete="current-password"
                  />
                </div>
              </div>

              {error && (
                <div className={styles.error} role="alert">
                  {error}
                </div>
              )}

              <button
                type="button"
                className={`${styles.authenticate} ${
                  authenticating ? styles.authenticating : ""
                }`}
                onClick={handleAuthenticate}
                disabled={authenticating}
              >
                <span>
                  {authenticating
                    ? "AUTHENTICATING..."
                    : "AUTHENTICATE"}
                </span>

                <span className={styles.arrow}>
                  {authenticating ? "•••" : "→"}
                </span>

                <span className={styles.scan} />
              </button>
            </>
          )}

          {authenticated && (
            <div className={styles.confirmation}>
              <div className={styles.confirmationRing}>
                <span>✓</span>
              </div>

              <div>
                <strong>IDENTITY CONFIRMED</strong>
                <small>SECURE SESSION ESTABLISHED</small>
              </div>
            </div>
          )}

          <div className={styles.panelFooter}>
            <span />
            <small>NARCIS SECURE NETWORK</small>
            <span />
          </div>

        <div className={styles.footer}>
          <span />
          <small>AUTHORIZED PERSONNEL ONLY</small>
          <span />
        </div>

    </main>
  );
}

from datetime import datetime

from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel
from sqlalchemy.orm import Session

from backend.auth.security import verify_password
from backend.database.database import get_db
from backend.database.models import Officer

router = APIRouter(
    prefix="/api/auth",
    tags=["Authentication"]
)


class LoginRequest(BaseModel):
    officer_id: str
    password: str


@router.post("/login")
def login(
    credentials: LoginRequest,
    db: Session = Depends(get_db)
):
    officer = (
        db.query(Officer)
        .filter(
            Officer.officer_id == credentials.officer_id
        )
        .first()
    )

    if officer is None:
        raise HTTPException(
            status_code=401,
            detail="Invalid Officer ID or password."
        )

    if officer.status != "Active":
        raise HTTPException(
            status_code=403,
            detail="Officer account is inactive."
        )

    if not verify_password(
        credentials.password,
        officer.password_hash
    ):
        raise HTTPException(
            status_code=401,
            detail="Invalid Officer ID or password."
        )

    officer.last_login = datetime.utcnow()
    db.commit()

    return {
        "authenticated": True,
        "officer": {
            "officer_id": officer.officer_id,
            "name": officer.name,
            "role": officer.role
        }
    }

def load_case_library():
    cases = []

    for case_file in sorted(CASE_LIBRARY.glob("case[0-9][0-9][0-9].json")):
        with case_file.open("r", encoding="utf-8-sig") as file:
            data = json.load(file)

        verification = data.get("_verification", {})

        cases.append({
            "case_id": case_file.stem.upper(),
            "drug_type": data.get("Drug_Type", "Unknown"),
            "quantity": data.get("Quantity", "Unknown"),
            "quantity_category": data.get(
                "Quantity_Category",
                "Unknown"
            ),
            "origin_state": data.get(
                "Origin_State",
                "Unknown"
            ),
            "destination_state": data.get(
                "Destination_State",
                "Unknown"
            ),
            "transport_mode": data.get(
                "Transport_Mode",
                "Unknown"
            ),
            "route_type": data.get(
                "Route_Type",
                "Unknown"
            ),
            "investigation_year": data.get(
                "Investigation_Year",
                "Unknown"
            ),
            "verification_status": (

                if verification.get("verified") is True
                else "Pending Verification"
            )
        })

    return cases


def load_evidence_package():

def load_final_report():
    report_path = OUTPUT_DIR / "final_investigation_report.json"

    if not report_path.exists():
        return None

    with report_path.open("r", encoding="utf-8-sig") as file:
        return json.load(file)


def identify_analyzed_case(evidence, cases):
    if not evidence:
        return None

    evidence_case = evidence.get("case", {})
    drug_type = evidence_case.get("Drug_Type", "Unknown")
    quantity = evidence_case.get("Quantity", "Unknown")

    matches = [
        case
        for case in cases
        if case["drug_type"] == drug_type
        and case["quantity"] == quantity
    ]

    return matches[0] if matches else None


@router.get("/stats")
def dashboard_stats():
    cases = load_case_library()
    evidence = load_evidence_package()
    report = load_final_report()

    verified_cases = [
        case
        for case in cases
        if case["verification_status"] == "Verified"
    ]

    analyzed_case = identify_analyzed_case(
        evidence,
        cases
    )

    analyzed_cases = 0
    high_priority = 0
    medium_priority = 0
    low_priority = 0

    if evidence:
        analyzed_cases = 1

        prediction = (
            evidence
            .get("ml_prediction", {})
            .get("prediction")
        )

        if prediction == "High":
            high_priority = 1

        if prediction == "Medium":
            medium_priority = 1

        if prediction == "Low":
            low_priority = 1

    return {
        "total_cases": len(cases),
        "active_cases": len(cases),
        "verified_cases": len(verified_cases),
        "pending_verification": (
            len(cases) - len(verified_cases)
        ),
        "analyzed_cases": analyzed_cases,
        "high_priority": high_priority,
        "medium_priority": medium_priority,
        "low_priority": low_priority,
        "reports_generated": (
            1 if report else 0
        ),
        "analyzed_case_id": (
            analyzed_case["case_id"]
            if analyzed_case
            else None
        )
    }

from passlib.context import CryptContext

pwd_context = CryptContext(
    schemes=["bcrypt"],
    deprecated="auto"
)


def hash_password(password: str) -> str:
    return pwd_context.hash(password)


def verify_password(
    plain_password: str,
    password_hash: str
) -> bool:
    return pwd_context.verify(
        plain_password,
        password_hash
    )
